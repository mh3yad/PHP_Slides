دوال الـ Sort · تطبيق

مقارنة عملية بالكود بين asort() و ksort()

تطبيق الدالتين على نفس الـ Array لملاحظة الفارق الدقيق في النتيجة:

asort_values.php 
$prices = [
    "شاشة"  => 5000,
    "ماوس"   => 300,
    "كيبورد" => 800
];

// Sort بالقيم تصاعدياً (من الأرخص للأغلى)
asort($prices);

/* الناتج:
ماوس   => 300
كيبورد => 800
شاشة   => 5000
*/
ksort_keys.php 
$codes = [
    "US" => "أمريكا",
    "EG" => "مصر",
    "SA" => "السعودية"
];

// Sort بالمفاتيح تصاعدياً (أبجدياً EG, SA, US)
ksort($codes);

/* الناتج:
EG => مصر
SA => السعودية
US => أمريكا
*/
