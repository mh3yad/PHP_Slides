تطبيق 1: نظام تسجيل دخول وحماية الصفحات

بناء جدار حماية (Auth Guard) يمنع الزوار غير المسجلين من دخول لوحة التحكم ويحولهم لصفحة الدخول:

auth_guard.php 
// يتم تضمين هذا الملف في أول كل صفحة محمية
session_start();

if (empty($_SESSION['logged_in'])) {
    // المستخدم غير مسجل، حوله لصفحة الدخول
    header("Location: login.php");
    exit; // ضروري لإيقاف تنفيذ الكود!
}
dashboard.php 
<?php
require 'auth_guard.php';
?>
<h1>لوحة تحكم المشرف</h1>
<p>أهلاً <?= $_SESSION['username'] ?>!</p>
<a href="logout.php">تسجيل الخروج 🚪</a>
