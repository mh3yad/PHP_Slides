تطبيق 4: رسائل التنبيه لمرة واحدة (Flash Messages)

إظهار رسالة تنبيه للمستخدم (مثل «تم حفظ البيانات بنجاح!») تختفي تلقائياً عند تحديث الصفحة أو الانتقال:

save_action.php 
session_start();

// تنفيذ عملية الحفظ بنجاح
$_SESSION['flash_msg']  = "تم حفظ التعديلات بنجاح! ✨";
$_SESSION['flash_type'] = "success";

// إعادة توجيه لصفحة العرض
header("Location: view.php");
exit;
view.php 
<?php
session_start();

// عرض الرسالة ثم مسحها فوراً
if (isset($_SESSION['flash_msg'])) {
    echo "<div class='alert'>{$_SESSION['flash_msg']}</div>";
    
    // مسح الرسالة لتختفي في التحديث القادم!
    unset($_SESSION['flash_msg']);
}
?>
